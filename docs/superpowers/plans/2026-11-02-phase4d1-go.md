```go
type syncStatus struct {
	ops.StoreStatus
	NextFetch *time.Time `json:"next_fetch,omitempty" doc:"When the loop fetches next; absent before its first sync"`
	PushAt    *time.Time `json:"push_at,omitempty" doc:"When the background sync pushes the writes not pushed yet (pending_since + sync.push_interval); absent when nothing waits"`
}

// newSyncStatus is st with when the loop fetches next and when the writes
// waiting are pushed (push is sync.push_interval).
func newSyncStatus(st ops.StoreStatus, l *Loop, push time.Duration) syncStatus {
	v := syncStatus{StoreStatus: st}
	if t, ok := l.NextFetch(st.Store); ok {
		v.NextFetch = &t
	}
	if st.PendingSince != nil {
		t := st.PendingSince.Add(push)
		v.PushAt = &t
	}
	return v
}
```

```go
package ops

import (
	"maps"
	"slices"

	"github.com/kukv/hedgehug/internal/gitrepo"
	"github.com/kukv/hedgehug/internal/index"
	"github.com/kukv/hedgehug/internal/mount"
)

// People are the names the GUI offers after an @: the assignees, watchers and
// comment authors of the tasks, and the Hedgehug-Author of the commits. They
// are those of the store the session selects, or of every store; sorted, each
// once. A name that cannot be written as a mention (@login or @host/login,
// as index.Mentions reads it) is left out: the GUI writes the name after an @.
func (ss *Session) People() ([]string, error) {
	set, err := ss.listSet()
	if err != nil {
		return nil, err
	}
	stores := set.Stores
	if ss.StoreName != "" {
		m, _ := set.Get(ss.StoreName)
		stores = []*mount.Mounted{m}
	}
	seen := map[string]bool{}
	add := func(name string) {
		if name != "" && slices.Equal(index.Mentions("@"+name), []string{name}) {
			seen[name] = true
		}
	}
	for _, m := range stores {
		ix, err := m.Index()
		if err != nil {
			return nil, classify(err)
		}
		for _, e := range ix.Tasks() {
			for _, n := range e.Assignees {
				add(n)
			}
			for _, n := range e.Watchers {
				add(n)
			}
			for _, c := range ix.Comments(e.ULID) {
				add(c.Author)
			}
		}
		var entries []gitrepo.LogEntry
		var logErr error
		if err := m.With(func(*index.Index) { entries, logErr = m.Store.Repo().Log(nil, 0) }); err != nil {
			return nil, classify(err)
		}
		if logErr != nil {
			return nil, logErr
		}
		for _, en := range entries {
			a, _ := en.Trailer("Hedgehug-Author")
			add(a)
		}
	}
	return slices.AppendSeq([]string{}, slices.Values(slices.Sorted(maps.Keys(seen)))), nil
}
```

`internal/ops/showat.go`（新規）。コミットの読み方は `RestoreAt` と同じ（`commitAt` と `ReadAt`）で、何も書かない。ID・旧 ID・ワークスペースは今のもの（時点の ID は別のタスクのものだったかもしれないため）。`Problems`・`Comments`・`History`・`Reverse`・`ReferencedBy` は空にする。

```go
package ops

import (
	"cmp"
	"errors"
	"fmt"
	"io/fs"

	"github.com/kukv/hedgehug/internal/index"
	"github.com/kukv/hedgehug/internal/model"
)

// ShowAt is the task arg as it was at at (a time or a commit, as RestoreAt
// takes it), read from the commit and written nowhere. The id, the workspace
// and the problems are those of now; the comments, the history and the
// relations are left empty.
func (ss *Session) ShowAt(arg, at string) (ShowView, error) {
	ts, arg, err := ss.forTask(arg)
	if err != nil {
		return ShowView{}, err
	}
	if ts != ss {
		return ts.ShowAt(arg, at)
	}
	s, ix, err := ss.Open()
	if err != nil {
		return ShowView{}, err
	}
	e, err := ss.Resolve(ix, arg)
	if err != nil {
		return ShowView{}, err
	}
	commit, err := commitAt(s.Repo(), at, ss.location())
	if err != nil {
		return ShowView{}, err
	}
	data, err := s.Repo().ReadAt(commit, model.TaskPath(e.ULID))
	if errors.Is(err, fs.ErrNotExist) {
		return ShowView{}, newErr(KindNotFound, "restore_at.not_found", fmt.Errorf("%s did not exist at %s", cmp.Or(ix.DisplayID(e), e.ULID.String()), at), "at", at)
	}
	if err != nil {
		return ShowView{}, err
	}
	old, err := model.DecodeTask(data)
	if err != nil {
		return ShowView{}, err
	}
	old.ID, old.FormerIDs, old.Workspace = e.ID, e.FormerIDs, e.Workspace
	entry := index.EntryOf(old)
	v := ShowView{
		TaskView: TaskViewOf(ix, entry),
		Body:     old.Body,
		Watchers: OrEmpty(old.Watchers),
		Repos:    OrEmpty(old.Repos),
		Comments: []CommentView{},
		Reverse:  map[string][]Ref{},
		History:  []LogView{},
	}
	v.Problems = []Problem{}
	v.ReferencedBy = []Ref{}
	v.Checklist = ChecklistOf(entry)
	return v, nil
}
```
