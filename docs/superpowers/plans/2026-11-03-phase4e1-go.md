```go
// projectOf is the nearest ancestor of e of kind project, or nil. e itself is not looked at. An ancestor the index does not have, or
// a cycle of parents, ends the walk with nil.
func projectOf(ix *index.Index, e *index.Entry) *Ref {
	seen := map[model.ULID]bool{e.ULID: true}
	for p := e.Parent; p != nil; {
		if seen[*p] {
			return nil
		}
		seen[*p] = true
		a, ok := ix.Task(*p)
		if !ok {
			return nil
		}
		if a.Kind == projectKind {
			r := RefOf(ix, a.ULID)
			return &r
		}
		p = a.Parent
	}
	return nil
}
```

### Task 3: プロジェクトの上書き（scoped）の書き込み（Go）

利用者の決定 1 と細部 3。ある container（プロジェクトなど）の `schema.yaml` の `scoped` の上書きを、ops・CLI・API から読み書きできるようにする。大きいので 3a（schema と ops）・3b（CLI）・3c（API、文言、型の作り直し）に分け、3つのコミットにする。各 Step の完了条件はその Step の `go test` が通ること。

決めたこと（概要 plan の細部 3 を、次のとおりに具体化する）:

- 渡し方は ops では `schema.ScopedChange`（`Statuses *[]string`、`Transitions *Lists`、`Required *Lists`）。**nil はその部分を変えない。渡したが空の部分は、その部分を外して Store の値に戻す**。API の本文でも同じで、**省いた項目は変えず、空の配列・空のオブジェクトは外す。`null` は意味を持たせず、400 `request.invalid`**（OpenAPI の型が非 null のため、バリデーションで落ちる）。CLI は指定したフラグの部分だけを渡し、空の値（`--statuses ""`、`--transition ""`、`--required ""`）が「外す」になる
- 3つの部分（statuses・transitions・required）のそれぞれが「置き換え」の単位。transitions の中の1つの状態だけを足す・消す操作は作らない（`hh schema scope set` は渡した部分を丸ごと置き換える。画面は今の値を読んで全部を送る）
- 上書きが `statuses`・`workflow`・`templates` のどれも持たなくなったら、`scoped` の要素ごと消す。`templates` と、`schema.yaml` にある未知のキーは書き戻しで残る（`internal/model` の `Encoder` がファイルのノードに値を混ぜて書くため。テストで確かめる）
- 検査は `Schema.Check` と同じ規則。**書く前の `Check` に無かった問題を作る書き込みは 422 `scoped.invalid`（ops では `KindInvalid`）で拒否**する。すでに別の問題があるスキーマに書くと Tx が `store.SchemaError` で止めるので、それも同じ `scoped.invalid` にする
- 書き込みは1コミット、op は `schema`、`Hedgehug-Task` はそのプロジェクト。`undo` で戻せる（テストで確かめる）
- 対象は **container の role の種類の、削除されていないタスク**（プロジェクトに限らずエピックも）。`ScopedGet` と `ScopedRemove` は、削除済みのプロジェクトでも、タスクがもう無くても（ULID で指定すれば）使える。`doctor` が「`where` が存在しないか削除済みのプロジェクトを指す」と報告したあとに、上書きを外す手段が要るため

```go
package schema

import (
	"slices"

	"github.com/kukv/hedgehug/internal/model"
)

// ScopedChange is a change to the override of one project. A nil part is
// left as it is; a part that is given but empty is taken off, so the project
// has the store's value again.
type ScopedChange struct {
	Statuses    *[]string
	Transitions *Lists
	Required    *Lists
}

// ScopedFor returns the override of project.
func (s *Schema) ScopedFor(project model.ULID) (Scoped, bool) {
	if i := s.scopedIndex(project); i >= 0 {
		return s.Scoped[i], true
	}
	return Scoped{}, false
}

func (s *Schema) scopedIndex(project model.ULID) int {
	return slices.IndexFunc(s.Scoped, func(sc Scoped) bool {
		w, err := parseWhere(sc.Where)
		return err == nil && w == project
	})
}

// WithScoped returns a copy of s whose override of project is changed by ch.
// An override left with no statuses, no workflow and no templates is dropped.
func (s *Schema) WithScoped(project model.ULID, ch ScopedChange) *Schema {
	out := s.clone()
	i := out.scopedIndex(project)
	sc := Scoped{Where: "project:" + project.String()}
	if i >= 0 {
		sc = out.Scoped[i]
	}
	if ch.Statuses != nil {
		sc.Statuses = nil
		if len(*ch.Statuses) > 0 {
			sc.Statuses = slices.Clone(*ch.Statuses)
		}
	}
	if ch.Transitions != nil || ch.Required != nil {
		var w Workflow
		if sc.Workflow != nil {
			w = *sc.Workflow
		}
		if ch.Transitions != nil {
			w.Transitions = nilIfEmpty(ch.Transitions.clone())
		}
		if ch.Required != nil {
			w.Required = nilIfEmpty(ch.Required.clone())
		}
		sc.Workflow = &w
		if w.IsZero() {
			sc.Workflow = nil
		}
	}
	switch empty := sc.Statuses == nil && sc.Workflow == nil && sc.Templates == nil; {
	case i >= 0 && empty:
		out.Scoped = slices.Delete(out.Scoped, i, i+1)
	case i >= 0:
		out.Scoped[i] = sc
	case !empty:
		out.Scoped = append(out.Scoped, sc)
	}
	return out
}

// WithoutScoped returns a copy of s without the override of project.
func (s *Schema) WithoutScoped(project model.ULID) *Schema {
	out := s.clone()
	if i := out.scopedIndex(project); i >= 0 {
		out.Scoped = slices.Delete(out.Scoped, i, i+1)
	}
	return out
}

func nilIfEmpty(l Lists) Lists {
	if len(l) == 0 {
		return nil
	}
	return l
}

// NewProblems lists the problems of s that are not in before.
func (s *Schema) NewProblems(before []Problem) []Problem {
	var out []Problem
	for _, p := range s.Check() {
		if !slices.Contains(before, p) {
			out = append(out, p)
		}
	}
	return out
}
```

`internal/ops/scoped.go` を作る。`ScopedSet` は、タスクの解決・削除済みと container の検査をしてから、`UpdateCommit` の中で **ロックの下で読み直した `s.Schema()`** から次のスキーマを作る（Migrate と同じ。ほかの書き込みを失わない）。新しい問題があれば書かずに `scoped.invalid`。

```go
package ops

import (
	"errors"
	"fmt"
	"strings"

	"github.com/kukv/hedgehug/internal/index"
	"github.com/kukv/hedgehug/internal/model"
	"github.com/kukv/hedgehug/internal/schema"
	"github.com/kukv/hedgehug/internal/store"
)

// ScopedProject names the project of an override. ID and Title are empty
// for an override whose project is no longer in the store.
type ScopedProject struct {
	ULID  string `json:"ulid"`
	ID    string `json:"id,omitempty"`
	Title string `json:"title,omitempty"`
}

// ScopedView is the override of a project as hh schema scope prints it with
// --json and PUT and DELETE /api/schema/scoped/{project} return it. Scoped is
// left out when the project has no override; Commit is null when nothing was
// written.
type ScopedView struct {
	Project ScopedProject `json:"project"`
	Scoped  *ScopedDef    `json:"scoped,omitempty"`
	Commit  *string       `json:"commit"`
}

// ScopedResult is what ScopedGet, ScopedSet and ScopedRemove did. Scoped is
// the override as it is after the call, nil when there is none. Entry is nil
// when the project is no longer in the store. Index is the one the call read:
// read the names of tasks from it, not the schema.
type ScopedResult struct {
	Index     *index.Index
	Project   model.ULID
	Entry     *index.Entry
	Scoped    *schema.Scoped
	Committed bool
	Commit    string // hex hash of the commit made; "" when nothing was committed
}

// View is r as ScopedView.
func (r ScopedResult) View() ScopedView {
	v := ScopedView{Project: ScopedProject{ULID: r.Project.String()}}
	if r.Entry != nil {
		v.Project.ID, v.Project.Title = r.Index.DisplayID(r.Entry), r.Entry.Title
	}
	if r.Scoped != nil {
		d := ScopedDefOf(*r.Scoped)
		v.Scoped = &d
	}
	if r.Commit != "" {
		c := r.Commit
		v.Commit = &c
	}
	return v
}

// scopedProject resolves the project arg names. An override whose project is
// gone is still named by its ULID, so that it can be shown and removed.
func (ss *Session) scopedProject(ix *index.Index, arg string) (model.ULID, *index.Entry, error) {
	e, err := ss.Resolve(ix, arg)
	if err == nil {
		return e.ULID, e, nil
	}
	if KindOf(err) == KindNotFound {
		if u, perr := model.ParseULID(arg); perr == nil {
			if _, ok := ix.Schema().ScopedFor(u); ok {
				return u, nil, nil
			}
		}
	}
	return model.ULID{}, nil, err
}

// ScopedGet reads the override of the project arg names.
func (ss *Session) ScopedGet(arg string) (ScopedResult, error) {
	_, ix, err := ss.Open()
	if err != nil {
		return ScopedResult{}, err
	}
	ulid, e, err := ss.scopedProject(ix, arg)
	if err != nil {
		return ScopedResult{}, err
	}
	r := ScopedResult{Index: ix, Project: ulid, Entry: e}
	if sc, ok := ix.Schema().ScopedFor(ulid); ok {
		r.Scoped = &sc
	}
	return r, nil
}

// ScopedSet changes the override of the project arg names, in one commit
// (op schema): the parts ch gives are replaced, a part ch gives empty goes
// back to the store's value, and the rest is kept, with the templates and the
// keys schema.yaml has that hedgehug does not know. An override left with
// nothing is removed. The project must be a task of a kind whose role is
// container and not deleted (KindInvalid, scoped.not_container; KindConflict,
// task.deleted). A change that brings a problem schema.Check does not find in
// schema.yaml now is KindInvalid, scoped.invalid, with the problems as Args
// "problems"; nothing is written.
func (ss *Session) ScopedSet(arg string, ch schema.ScopedChange) (ScopedResult, error) {
	s, ix, err := ss.OpenForWrite()
	if err != nil {
		return ScopedResult{}, err
	}
	e, err := ss.Resolve(ix, arg)
	if err != nil {
		return ScopedResult{}, err
	}
	if err := NotDeleted(ix, e); err != nil {
		return ScopedResult{}, err
	}
	id := ix.DisplayID(e)
	if k, ok := ix.Schema().Kind(e.Kind); !ok || k.Role != schema.RoleContainer {
		return ScopedResult{}, newErr(KindInvalid, "scoped.not_container",
			fmt.Errorf("%s is a %s, which is not a container", id, e.Kind), "id", id, "kind", e.Kind)
	}
	return ss.scopedWrite(s, ix, e.ULID, e, func(sc *schema.Schema) *schema.Schema { return sc.WithScoped(e.ULID, ch) })
}

// ScopedRemove removes the override of the project arg names, templates
// included, in one commit (op schema). A project with no override is not an
// error; nothing is committed. The project may be deleted, and may be gone
// when arg is its ULID.
func (ss *Session) ScopedRemove(arg string) (ScopedResult, error) {
	s, ix, err := ss.OpenForWrite()
	if err != nil {
		return ScopedResult{}, err
	}
	ulid, e, err := ss.scopedProject(ix, arg)
	if err != nil {
		return ScopedResult{}, err
	}
	return ss.scopedWrite(s, ix, ulid, e, func(sc *schema.Schema) *schema.Schema { return sc.WithoutScoped(ulid) })
}

// scopedWrite writes the schema that next makes from the one the store has
// under its lock.
func (ss *Session) scopedWrite(s *store.Store, ix *index.Index, ulid model.ULID, e *index.Entry, next func(*schema.Schema) *schema.Schema) (ScopedResult, error) {
	var after *schema.Schema
	hash, err := ss.UpdateCommit(s, store.Op{Name: "schema", Task: ulid}, func(tx *store.Tx) error {
		cur := s.Schema()
		after = next(cur)
		if ps := after.NewProblems(cur.Check()); len(ps) > 0 {
			return scopedInvalid(ps)
		}
		return tx.PutSchema(after)
	})
	var se *store.SchemaError
	if errors.As(err, &se) {
		err = scopedInvalid(se.Problems)
	}
	if err != nil {
		return ScopedResult{}, err
	}
	r := ScopedResult{Index: ix, Project: ulid, Entry: e, Committed: !hash.IsZero(), Commit: commitHex(hash)}
	if sc, ok := after.ScopedFor(ulid); ok {
		r.Scoped = &sc
	}
	return r, nil
}

func scopedInvalid(ps []schema.Problem) error {
	msgs := make([]string, len(ps))
	for i, p := range ps {
		msgs[i] = p.String()
	}
	return newErr(KindInvalid, "scoped.invalid", fmt.Errorf("the override breaks the schema: %s", strings.Join(msgs, "; ")), "problems", msgs)
}
```

```go
// splitNames splits a comma separated list; an empty value is an empty list.
func splitNames(flag, v string) ([]string, error) {
	out := []string{}
	if v == "" {
		return out, nil
	}
	for _, n := range strings.Split(v, ",") {
		n = strings.TrimSpace(n)
		if n == "" {
			return nil, usageErrorf("%s: %q has an empty name", flag, v)
		}
		out = append(out, n)
	}
	return out, nil
}

// listsFlag reads the values of a repeatable flag of the form key=a,b. One
// empty value alone is an empty set of lists, which takes the part off.
func listsFlag(flag string, vals []string) (*schema.Lists, error) {
	l := schema.Lists{}
	if len(vals) == 1 && vals[0] == "" {
		return &l, nil
	}
	for _, v := range vals {
		key, rest, ok := strings.Cut(v, "=")
		key = strings.TrimSpace(key)
		if !ok || key == "" {
			return nil, usageErrorf("%s: %q is not of the form key=a,b", flag, v)
		}
		if _, dup := l.Get(key); dup {
			return nil, usageErrorf("%s: %q is given twice", flag, key)
		}
		names, err := splitNames(flag, rest)
		if err != nil {
			return nil, err
		}
		l = append(l, schema.List{Key: key, Values: names})
	}
	return &l, nil
}

// printScoped prints the override r is about. A write starts with a line
// that says what it did, done, or that it changed nothing, none; both are
// empty for show. With body the override as it is now follows.
func (a *app) printScoped(r ops.ScopedResult, done, none string, body bool) error {
	if a.jsonOut {
		return a.printJSON(r.View())
	}
	name := r.Project.String()
	if r.Entry != nil {
		name = r.Index.DisplayID(r.Entry) + "  " + r.Entry.Title
	}
	var b strings.Builder
	if done != "" {
		if !r.Committed {
			_, err := fmt.Fprintf(a.stdout, "%s %s\n", none, name)
			return err
		}
		fmt.Fprintf(&b, "%s the override of %s\n", done, name)
	}
	if body {
		writeScoped(&b, name, r.Scoped)
	}
	_, err := fmt.Fprint(a.stdout, b.String())
	return err
}

func writeScoped(b *strings.Builder, name string, sc *schema.Scoped) {
	if sc == nil {
		fmt.Fprintf(b, "%s: no override\n", name)
		return
	}
	fmt.Fprintf(b, "%s\n", name)
	if sc.Statuses != nil {
		fmt.Fprintf(b, "  statuses: %s\n", strings.Join(sc.Statuses, ", "))
	}
	if w := sc.Workflow; w != nil {
		writeLists(b, "transitions", w.Transitions)
		writeLists(b, "required", w.Required)
	}
	if sc.Templates != nil {
		b.WriteString("  templates: kept (they are edited in schema.yaml)\n")
	}
}

func writeLists(b *strings.Builder, heading string, l schema.Lists) {
	if len(l) == 0 {
		return
	}
	fmt.Fprintf(b, "  %s:\n", heading)
	for _, x := range l {
		fmt.Fprintf(b, "    %s: %s\n", x.Key, strings.Join(x.Values, ", "))
	}
}
```

```go
package server

import (
	"context"
	"errors"
	"net/http"
	"slices"

	"github.com/danielgtaylor/huma/v2"

	"github.com/kukv/hedgehug/internal/ops"
	"github.com/kukv/hedgehug/internal/schema"
)

// ScopedParam names the project of an override. It is exported because huma
// skips an embedded struct whose type is not.
type ScopedParam struct {
	StoreParam
	Project string `path:"project" doc:"The project: its ID or ULID"`
}

type scopedPutInput struct {
	ScopedParam
	Body struct {
		Statuses    *[]string            `json:"statuses,omitempty" doc:"The statuses to show; the default of each category has to be in it. Leave it out to keep what the override has; an empty list takes it off, so the store's statuses show"`
		Transitions *map[string][]string `json:"transitions,omitempty" doc:"Status to the statuses it may move to, on top of the store's. Leave it out to keep what the override has; an empty object takes it off"`
		Required    *map[string][]string `json:"required,omitempty" doc:"Status to the fields needed to enter it, on top of the store's. Leave it out to keep what the override has; an empty object takes it off"`
	}
}

type scopedOutput struct{ Body ops.ScopedView }

// listsOf is m as schema.Lists, in the order of the keys.
func listsOf(m *map[string][]string) *schema.Lists {
	if m == nil {
		return nil
	}
	keys := make([]string, 0, len(*m))
	for k := range *m {
		keys = append(keys, k)
	}
	slices.Sort(keys)
	l := make(schema.Lists, 0, len(keys))
	for _, k := range keys {
		vals := (*m)[k]
		if vals == nil {
			vals = []string{}
		}
		l = append(l, schema.List{Key: k, Values: vals})
	}
	return &l
}

func (s *Server) registerScoped() {
	huma.Register(s.api, huma.Operation{
		OperationID: "put-scoped", Method: http.MethodPut, Path: "/api/schema/scoped/{project}",
		Summary: "Change the override of a project, as hh schema scope set", Tags: []string{"schema"},
	}, func(_ context.Context, in *scopedPutInput) (*scopedOutput, error) {
		b := in.Body
		if b.Statuses == nil && b.Transitions == nil && b.Required == nil {
			return nil, badRequest(errors.New("give statuses, transitions or required"))
		}
		r, err := s.session(in.Store).ScopedSet(in.Project, schema.ScopedChange{
			Statuses: b.Statuses, Transitions: listsOf(b.Transitions), Required: listsOf(b.Required),
		})
		if err != nil {
			return nil, problemOf(err)
		}
		return &scopedOutput{r.View()}, nil
	})

	huma.Register(s.api, huma.Operation{
		OperationID: "delete-scoped", Method: http.MethodDelete, Path: "/api/schema/scoped/{project}",
		Summary: "Remove the override of a project, as hh schema scope rm", Tags: []string{"schema"},
	}, func(_ context.Context, in *ScopedParam) (*scopedOutput, error) {
		r, err := s.session(in.Store).ScopedRemove(in.Project)
		if err != nil {
			return nil, problemOf(err)
		}
		return &scopedOutput{r.View()}, nil
	})
}
```
