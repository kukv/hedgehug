## この plan で決めたこと（概要の細部への足し分と直し）

1. **poll の手放し（細部 11 を直す）**: `poller.Release` だけでは、`StoreRemove` が config を書く前に poll が Store を開き直せる。`Poller.ReleaseFor(path, fn)` を足し、repo を手放してから `fn`（`StoreRemove`）が終わるまで `pollMu` を持つ。`Release` と処理を共有する（Task 2）。`DELETE /api/stores/{name}` は `ReleaseFor` の前に config の Entry を読み、`StoreRemove` がもう一度読む（読みが2回になる）が、`ReleaseFor` の外の1回は手放す対象のパスを知るためだけで、間に config が変わっても次の poll が立て直すので、そのままにする。消せずに失敗したときは、poller が同じ Store を黙って開き直す（`restored`。再構築の通知は出さない）

```go
// SetStoreLook changes how the GUI shows the store name: its color and
// whether it is hidden. A nil argument is left as it is; an empty color and
// hidden=false remove their keys. The color must be one of StoreColors.
// Save writes it.
func (c *Config) SetStoreLook(name string, color *string, hidden *bool) error {
	i := slices.IndexFunc(c.Stores, func(s StoreEntry) bool { return s.Name == name })

		return fmt.Errorf("no store is named %q", name)
```

```go
func noSuchStore(c *config.Config, name string) error {
	names := make([]string, len(c.Stores))
	for i, s := range c.Stores {
		names[i] = s.Name
	}
	msg := fmt.Sprintf("no store is named %q", name)
	if len(names) > 0 {

// StoreSet changes how the GUI shows the store name, as hh store set: its
// color (one of config.StoreColors; "" gives it back to the GUI to pick) and
// whether it is hidden. A nil argument is left as it is. config.yaml is not
// written when nothing changes. A color that is not one is KindInvalid,
// "store.bad_color"; a name that is no store is "store.not_found" as for
// StoreDefault. It returns the store as it is now.
func (ss *Session) StoreSet(name string, color *string, hidden *bool) (StoreView, error) {
	if color != nil && *color != "" && !slices.Contains(config.StoreColors, *color) {
		return StoreView{}, newErr(KindInvalid, "store.bad_color",
			fmt.Errorf("%q is not a store color (colors: %s)", *color, strings.Join(config.StoreColors, ", ")),
			"value", *color, "candidates", slices.Clone(config.StoreColors))
	}
	c, _, err := ss.Config()
	if err != nil {
		return StoreView{}, err
	}
	i := slices.IndexFunc(c.Stores, func(e config.StoreEntry) bool { return e.Name == name })
	if i < 0 {
		return StoreView{}, noSuchStore(c, name)
	}
	cur := c.Stores[i]
	if (color != nil && *color != cur.Color) || (hidden != nil && *hidden != cur.Hidden) {
		if err := ss.updateConfig(c, func(c *config.Config) error {
			if !storeExists(c, name) { // another writer may have removed it since the read
				return noSuchStore(c, name)
			}
			return c.SetStoreLook(name, color, hidden)
		}); err != nil {
			return StoreView{}, err
		}
	}
	i = slices.IndexFunc(c.Stores, func(e config.StoreEntry) bool { return e.Name == name })
	return ss.storeView(c, c.Stores[i]), nil
}
```

```go
// drainEvents reads what the hub has put in ch, without waiting. A poll
// publishes while it runs, so what a poll made is in ch when it returns.
func drainEvents(ch <-chan sseEvent) []sseEvent {
	var out []sseEvent
	for {
		select {
		case e, ok := <-ch:
			if !ok {
				return out
			}
			out = append(out, e)
		default:
			return out
		}
	}
}

// countStoresChanged is how many stores.changed are in evs.
func countStoresChanged(evs []sseEvent) int {
	n := 0
	for _, e := range evs {
		if _, ok := e.(storesChanged); ok {
			n++
		}
	}
	return n
}
```

```go
package server

// storesChanged is the event that the stores of config.yaml changed: one came
// or went, or its path, color, visibility or account did, or the accounts or
// the default store did. It has no body; the GUI reads GET /api/stores again.
type storesChanged struct{}

// storesWritten is what a write to the stores or the accounts of config.yaml
// through the API does afterwards: it tells the clients, makes the poller
// look now (so that a store that came is opened, and one that went is let
// go), and wakes the sync loop (a store that came is synced). The poller
// tells the clients too when it finds the change itself, so a client may get
// the event twice; reading the stores again is idempotent.
func (s *Server) storesWritten() {
	s.hub.Publish(storesChanged{})
	s.poller.Kick()
	s.loop.Wake()
}
```

```go
func (p *Poller) Release(path string) {
	p.pollMu.Lock()
	defer p.pollMu.Unlock()
	p.releaseLocked(path)
}

// releaseLocked is Release with pollMu held. It returns the names of the
// stores it let go of.
func (p *Poller) releaseLocked(path string) []string {
	var gone []*polled
	var names []string
	p.mu.Lock()
	for name, st := range p.stores {
		if st.path != path {
			continue
		}
		gone = append(gone, st)
		names = append(names, name)
		delete(p.stores, name)
		if p.released == nil {
			p.released = map[string]bool{}
		}
		p.released[name] = true
	}
	p.mu.Unlock()
	// The build of the history index reads the files under the directory
	// through a handle of its own, so the directory must not be renamed or
	// removed until it is over; the wait is made without p.mu, so hello does
	// not wait.
	for _, st := range gone {
		st.stopHistory()
		_ = st.s.Repo().Close()
	}
	return names
}

// ReleaseFor lets go of the store at path as Release does, and runs fn (the
// removal of the store: its directory goes) before any poll can look again,
// so that no poll opens the store again while its config entry is still
// there. When fn succeeds the store is gone for good, so it is no longer
// marked as swapped for a rebuild (a store of that name that comes later is
// new). fn must not wait for a poll.
func (p *Poller) ReleaseFor(path string, fn func() error) error {
	p.pollMu.Lock()
	defer p.pollMu.Unlock()
	names := p.releaseLocked(path)
	err := fn()
	if err == nil {
		p.mu.Lock()
		for _, n := range names {
			delete(p.released, n)
		}
		p.mu.Unlock()
	}
	return err
}
```

```go
func TestStoreDeleteAPI(t *testing.T) {
	t.Parallel()
	ts, ch := pollServer(t)
	bare := filepath.Join(t.TempDir(), "team.git")
	// A store with a remote it pushed to can be removed with its clone.
	if _, err := ts.session().StoreInit(usecase.StoreInitInput{Name: "team", Prefix: "TM", Remotes: []string{bare}}); err != nil {
		t.Fatal(err)
	}
	// And one with no remote cannot, unless the clone is kept.
	ts.call("POST", "/api/stores", map[string]any{"name": "local", "prefix": "LC"}, 201, nil)
	held := func(name string) bool {
		ts.srv.poller.mu.Lock()
		defer ts.srv.poller.mu.Unlock()
		return ts.srv.poller.stores[name] != nil
	}
	eventually(t, "the poller holds team and local", func() bool { return held("team") && held("local") })
	drainEvents(ch) // what the poller told of the two stores that came

	rows := ts.storeRows()
	var teamPath, localPath string
	for _, r := range rows {
		switch r.Name {
		case "team":
			teamPath = r.Path
		case "local":
			localPath = r.Path
		}
	}

	// Nothing is lost by the refusal: the store stays, held by the poller.
	if p := ts.problem("DELETE", "/api/stores/local", nil, 409); p.Type != "store.unpushed" || p.Args["name"] != "local" {
		t.Errorf("unpushed: %+v", p)
	}
	if _, err := os.Stat(localPath); err != nil {
		t.Errorf("a refused removal removed the clone: %v", err)
	}
	ts.problem("DELETE", "/api/stores/nope", nil, 404)

	// keep=true takes the store out of the config and leaves the clone.
	var res usecase.StoreRemoveResult
	ts.call("DELETE", "/api/stores/local?keep=true", nil, 200, &res)
	if res.Name != "local" || res.Path != localPath || res.Deleted {
		t.Errorf("keep = %+v", res)
	}
	if _, err := os.Stat(localPath); err != nil {
		t.Errorf("keep=true removed the clone: %v", err)
	}
	waitEvent[storesChanged](t, ch, nil)

	// The default removes the clone, after the poller let go of it.
	ts.call("DELETE", "/api/stores/team", nil, 200, &res)
	if res.Name != "team" || res.Path != teamPath || !res.Deleted {
		t.Errorf("delete = %+v", res)
	}
	if _, err := os.Stat(teamPath); !os.IsNotExist(err) {
		t.Errorf("the clone is left: %v", err)
	}
	eventually(t, "the poller to let go of team", func() bool { return !held("team") })
	waitEvent[storesChanged](t, ch, nil)
	if rows := ts.storeRows(); len(rows) != 1 || rows[0].Name != "me" {
		t.Errorf("stores = %+v", rows)
	}
}
```

```go
func (s *Server) registerStores() {
	huma.Register(s.api, huma.Operation{
		OperationID: "add-store", Method: http.MethodPost, Path: "/api/stores/add", DefaultStatus: http.StatusCreated,
		Summary: "Set up a store that others made by cloning it, as hh store add", Tags: []string{"stores"},
	}, func(_ context.Context, in *storeAddInput) (*storeViewOutput, error) {
		b := in.Body
		ss := s.session("")
		defer ss.Close()
		r, err := ss.StoreAdd(b.Name, b.Remotes, "", b.Account)
		if err != nil {
			return nil, problemOf(err)
		}
		v, err := storeViewOf(ss, r.Name)
		if err != nil {
			return nil, problemOf(err)
		}
		s.storesWritten()
		return &storeViewOutput{Body: v}, nil
	})

	huma.Register(s.api, huma.Operation{
		OperationID: "remove-store", Method: http.MethodDelete, Path: "/api/stores/{name}",
		Summary: "Take a store out of the config and delete its clone, as hh store remove", Tags: []string{"stores"},
	}, func(_ context.Context, in *storeRemoveInput) (*storeRemoveOutput, error) {
		ss := s.session(in.Name)
		defer ss.Close()
		var res usecase.StoreRemoveResult
		remove := func() (err error) {
			res, err = ss.StoreRemove(in.Name, in.Keep)
			return err
		}
		var err error
		// The poller holds the store open; the clone is deleted, so it lets go
		// first, and no poll opens it again until the removal is over. With
		// keep the clone stays, and the poller lets go when the store is gone
		// from the config. A name that is no store is StoreRemove's to refuse.
		if _, e, eerr := ss.Entry(); eerr == nil && !in.Keep {
			err = s.poller.ReleaseFor(e.Path, remove)
		} else {
			err = remove()
		}
		if err != nil {
			return nil, problemOf(err)
		}
		s.storesWritten()
		return &storeRemoveOutput{Body: res}, nil
	})

	huma.Register(s.api, huma.Operation{
		OperationID: "set-default-store", Method: http.MethodPut, Path: "/api/stores/default",
		Summary: "Make a store the one hedgehug picks without --store, as hh store default", Tags: []string{"stores"},
	}, func(_ context.Context, in *storeDefaultInput) (*storeDefaultOutput, error) {
		ss := s.session("")
		defer ss.Close()
		name, err := ss.StoreDefault(in.Body.Name)
		if err != nil {
			return nil, problemOf(err)
		}
		s.storesWritten()
		out := &storeDefaultOutput{}
		out.Body.Name = name
		return out, nil
	})

	huma.Register(s.api, huma.Operation{
		OperationID: "update-store", Method: http.MethodPatch, Path: "/api/stores/{name}",
		Summary: "Set the color or the visibility of a store on this machine, as hh store set", Tags: []string{"stores"},
	}, func(_ context.Context, in *storePatchInput) (*storeViewOutput, error) {
		b := in.Body
		if b.Color == nil && b.Hidden == nil {
			return nil, newProblem(http.StatusBadRequest, "store.set_args", "give color or hidden")
		}
		ss := s.session("")

		v, err := ss.StoreSet(in.Name, b.Color, b.Hidden)
		if err != nil {
			return nil, problemOf(err)
		}
		s.storesWritten()
		return &storeViewOutput{Body: v}, nil
	})
}
```

```go
// wait waits d, and reports false when ctx ended first. Sleep (the tests'
// clock) is used as it is; the cancel is looked at once it returns.
func (o Options) wait(ctx context.Context, d time.Duration) bool {
	if o.Sleep != nil {
		o.Sleep(d)
		return ctx.Err() == nil
	}
	t := time.NewTimer(d)
	defer t.Stop()
	select {
	case <-ctx.Done():
		return false
	case <-t.C:
		return true
	}
}
```

```go
// memKeyring is a keychain in memory.
type memKeyring struct {
	mu sync.Mutex
	m  map[string]string
}

func newMemKeyring() *memKeyring { return &memKeyring{m: map[string]string{}} }

func (k *memKeyring) Get(_, user string) (string, error) {
	k.mu.Lock()
	defer k.mu.Unlock()
	v, ok := k.m[user]
	if !ok {
		return "", auth.ErrNotFound
	}
	return v, nil
}

func (k *memKeyring) Set(_, user, secret string) error {
	k.mu.Lock()
	defer k.mu.Unlock()
	k.m[user] = secret
	return nil
}

func (k *memKeyring) Delete(_, user string) error {
	k.mu.Lock()
	defer k.mu.Unlock()
	if _, ok := k.m[user]; !ok {
		return auth.ErrNotFound
	}
	delete(k.m, user)
	return nil
}

func (k *memKeyring) has(user string) (string, bool) {
	k.mu.Lock()
	defer k.mu.Unlock()
	v, ok := k.m[user]
	return v, ok
}

}
```

```go
		ss := s.session("")

		out, err := ss.AuthStatus()

			return nil, problemOf(err)
		}

		}
```

```go
package server

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"net/http"

	"time"

// deviceRun is the device flow that is running.
type deviceRun struct {
	id     string
	cancel context.CancelFunc
	done   chan struct{} // closed by end
}

// deviceFlow holds the one device flow the server runs at a time.
type deviceFlow struct {
	mu      sync.Mutex
	current *deviceRun
}

// begin starts a run on a context of parent, unless one is running.
func (d *deviceFlow) begin(parent context.Context) (*deviceRun, context.Context, bool) {
	d.mu.Lock()
	defer d.mu.Unlock()
	if d.current != nil {
		return nil, nil, false
	}
	ctx, cancel := context.WithCancel(parent)
	b := make([]byte, 8)
	_, _ = rand.Read(b)
	r := &deviceRun{id: hex.EncodeToString(b), cancel: cancel, done: make(chan struct{})}
	d.current = r
	return r, ctx, true
}

// end clears r, before its event is published: a screen that starts the next
// flow on the event is not told 409.
func (d *deviceFlow) end(r *deviceRun) {
	d.mu.Lock()
	if d.current == r {
		d.current = nil
	}
	d.mu.Unlock()
	r.cancel()
	close(r.done)
}

// get is the run with the id, or nil.
func (d *deviceFlow) get(id string) *deviceRun {
	d.mu.Lock()
	defer d.mu.Unlock()
	if d.current != nil && d.current.id == id {
		return d.current
	}
	return nil
}

// wait cancels the run, if any, and waits until it ended.
func (d *deviceFlow) wait() {
	d.mu.Lock()
	r := d.current
	d.mu.Unlock()
	if r != nil {
		r.cancel()
		<-r.done
	}
}

func (s *Server) registerDevice() {
	huma.Register(s.api, huma.Operation{
		OperationID: "start-device-login", Method: http.MethodPost, Path: "/api/auth/device", DefaultStatus: http.StatusCreated,
		Summary: "Start the device flow, as hh auth login [<name>] [--host]; the server polls, and the event auth.device says how it ended", Tags: []string{"accounts"},
	}, func(ctx context.Context, in *deviceStartInput) (*deviceStartOutput, error) {
		run, rctx, ok := s.device.begin(s.base)
		if !ok {
			return nil, newProblem(http.StatusConflict, "auth.device_running",
				"a device flow is running; wait for it, or cancel it with DELETE /api/auth/device/{id}")
		}
		started := make(chan deviceStart, 1)
		go s.runDevice(run, rctx, in.Body.Name, in.Body.Host, started)
		select {
		case st := <-started:
			if st.prob != nil {
				return nil, st.prob
			}
			return &deviceStartOutput{Body: deviceStartBody{
				ID: run.id, UserCode: st.code.UserCode, VerificationURI: st.code.VerificationURI,
				ExpiresAt: s.opt.Now().Add(time.Duration(st.code.ExpiresIn) * time.Second),
			}}, nil
		case <-ctx.Done(): // the client went away before the code: nobody will enter it
			run.cancel()
			return nil, newProblem(http.StatusRequestTimeout, "request.canceled", "the request ended before the code was shown")
		}
	})

	huma.Register(s.api, huma.Operation{
		OperationID: "cancel-device-login", Method: http.MethodDelete, Path: "/api/auth/device/{id}",
		Summary: "Stop the device flow; it ends with the event auth.device (error, auth.canceled)", Tags: []string{"accounts"},
	}, func(ctx context.Context, in *deviceCancelInput) (*struct{}, error) {
		run := s.device.get(in.ID)
		if run == nil {
			return nil, newProblem(http.StatusNotFound, "auth.device_not_found", "no device flow "+in.ID+" is running")
		}
		run.cancel()
		select {
		case <-run.done:
		case <-ctx.Done():
		}
		return nil, nil
	})
}

// runDevice runs the login of one device flow and tells how it ended: the
// request that started it before the code was shown, the event after.
func (s *Server) runDevice(run *deviceRun, ctx context.Context, name, host string, started chan<- deviceStart) {
	ss := s.session("")
	defer ss.Close()
	shown := false
	res, err := ss.AuthLogin(usecase.AuthLoginInput{
		Name: name, Host: host, HostGiven: host != "", Context: ctx,
		OnCode: func(c usecase.DeviceCode) {
			shown = true
			started <- deviceStart{code: c}
		},
	})
	s.device.end(run)
	switch {
	case err != nil && !shown:
		started <- deviceStart{prob: problemOf(err)}
	case err != nil:
		e := usecase.AsError(err)
		code := e.Code
		if code == "" {
			code = "internal"
		}
		s.hub.Publish(authDeviceEvent{ID: run.id, State: "error", Code: code, Message: err.Error()})
	case res.Uses != "": // an account of gh or env:VAR: no code was asked for
		p := newProblem(http.StatusUnprocessableEntity, "auth.no_device_flow",
			"account "+res.Account+" takes its token from "+res.Uses+"; Hedgehug has nothing to log in")
		p.Args = map[string]any{"account": res.Account, "uses": res.Uses}
		started <- deviceStart{prob: p}
	default:
		s.pinnedStores(res.Pinned)
		s.hub.Publish(authDeviceEvent{ID: run.id, State: "done", Account: res.Account})
	}
}
```

```go
package usecase

import (
```

```go
// MovePlanView is a MovePlan as the GUI shows it: structured, where MoveView
// is the lines the CLI prints. It holds no content of the tasks (not their
// text, comments or worklogs), and it is what the user agrees to: the same
// view, made again under the locks, must equal it (MoveStoreInput.ConfirmedView).
type MovePlanView struct {
	From      string            `json:"from"`
	To        string            `json:"to"`
	Workspace MoveWorkspaceView `json:"workspace" doc:"The workspace of the destination that the root task goes into"`
	Parent    *MoveTaskView     `json:"parent,omitempty" doc:"The parent in the destination, when one was given; id is as the destination shows it"`
	Tasks     []MoveTaskView    `json:"tasks" nullable:"false" doc:"The root first, every parent before its children; id is as the source shows it"`
	Unlinked  []MoveUnlinkView  `json:"unlinked" nullable:"false" doc:"The references the move cuts"`
	Hidden    []string          `json:"hidden" nullable:"false" doc:"Accounts that can read the source and not the destination; empty when either list is unknown"`
	Scoped    *MoveScopedView   `json:"scoped,omitempty" doc:"Present when a project moved has a scoped schema"`
}

// MoveWorkspaceView names a workspace of the destination.
type MoveWorkspaceView struct {
	ULID   model.ULID `json:"ulid"`
	Name   string     `json:"name"`
	Prefix string     `json:"prefix"`
}

// MoveRefView names a task of the source.
type MoveRefView struct {
	ULID model.ULID `json:"ulid"`
	ID   string     `json:"id"`
}

// MoveTargetView names what a reference points at: a task, a sprint or a release.
type MoveTargetView struct {
	ULID  model.ULID `json:"ulid"`
	Label string     `json:"label" doc:"The ID of a task, the name of a sprint or a release; the ULID when there is none"`
}

// MoveUnlinkView is a reference the move cuts.

// GUIView is the plan as the GUI shows it.
func (p MovePlan) GUIView() MovePlanView {
	v := MovePlanView{
		From: p.From, To: p.To, Workspace: p.Workspace, Parent: p.Parent,
		Tasks:    make([]MoveTaskView, len(p.Tasks)),
		Unlinked: make([]MoveUnlinkView, len(p.Unlinked)),
		Hidden:   slices.Clone(p.Hidden),
	}
	if v.Hidden == nil {
		v.Hidden = []string{}
	}
	for i, e := range p.Tasks {
		v.Tasks[i] = MoveTaskView{ULID: e.ULID, ID: p.Label(e.ULID), Title: e.Title}
	}
	for i, u := range p.Unlinked {
		v.Unlinked[i] = MoveUnlinkView{
			Task:   MoveRefView{ULID: u.Task, ID: p.Label(u.Task)},
			Field:  u.Field,
			Target: MoveTargetView{ULID: u.Target, Label: p.Label(u.Target)},
		}
	}
	if p.Scoped != nil {
		v.Scoped = &MoveScopedView{Required: p.ScopedChoice == "", Choice: p.ScopedChoice, Dropped: []string{}}
		v.Scoped.Dropped = append(v.Scoped.Dropped, p.Scoped.Dropped...)
	}
	return v
}

// canonical gives the lists that are empty the same form, so that a view
// that went through JSON compares equal to the one made here.
func (v MovePlanView) canonical() MovePlanView {
	if v.Tasks == nil {
		v.Tasks = []MoveTaskView{}
	}
	if v.Unlinked == nil {
		v.Unlinked = []MoveUnlinkView{}
	}
	if v.Hidden == nil {
		v.Hidden = []string{}
	}
	if v.Scoped != nil && v.Scoped.Dropped == nil {
		s := *v.Scoped
		s.Dropped = []string{}
		v.Scoped = &s
	}
	return v
}

// samePlanView reports whether a and b are the same plan: the same tasks,
// where they go, the references cut, who loses sight and what the scoped

func samePlanView(a, b MovePlanView) bool {
	return reflect.DeepEqual(a.canonical(), b.canonical())
}
```

```go
		OperationID: "move-store", Method: http.MethodPost, Path: "/api/tasks/{id}/move-store",
		Summary: "Move the task and the tasks under it to another store, as hh mv --store after its question", Tags: []string{"tasks"},
	}, func(_ context.Context, in *moveStoreInput) (*moveStoreOutput, error) {
		mi := in.Body.input(in.ID)
		// Without a confirmed plan Yes is false, and MoveStore answers
		// confirm.required, as the CLI without --yes.
		mi.Yes, mi.ConfirmedView = in.Body.Confirmed != nil, in.Body.Confirmed
		ss := s.session(in.Store)
		defer ss.Close()
		res, err := ss.MoveStore(mi)
		if err != nil {
			return nil, problemOf(err)
		}
		s.poller.Kick()
		return &moveStoreOutput{Body: res.View()}, nil
	})
}
```
